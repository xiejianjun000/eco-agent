## 1. 数据结构

```
// 文书文档
DocState {
  docId: string
  filename: string          // "金竹山矿业_行政处罚决定书_草稿.docx"
  format: "docx" | "xlsx" | "pdf"

  revision: {
    type: "insert" | "delete" | "modify"
    by: "human" | "ai"
    oldText?: string        // 修改前的文本（hover 显示）
  } | null
  locked: boolean           // 当前谁在编辑（协作锁）
  lockedBy?: string
}

Annotation {
  id: string
  paragraphId: string       // 绑定的段落
  author: string            // 李建国 / 文书成（AI）
  authorType: "human" | "ai"
  text: string
  time: datetime
  status: "open" | "resolved"
  replies: AnnotationReply[]
}

AnnotationReply {
  author: string
  authorType: "human" | "ai"
  text: string
  time: datetime
}

SyncUser {
  name: string              // "李建国" | "文书成（AI）"
  type: "human" | "ai"
  status: "idle" | "writing_segment_N" | "reviewing"
  lastActive: datetime
}
```

## 3. 三模式切换

```
function switchMode(mode: "read" | "edit" | "review"):
  currentMode = mode

  switch mode:
    case "read":
      // 只渲染正文，所有段落只读
      disableAllEdits()
      toolbar = [缩放]
      showAnnotations = false

    case "edit":
      // 可编辑模式：人类可直接修改
      enableTextEditing()
      toolbar = [字号, 加粗, 列表, 插入引用条文]
      showAnnotations = false
      // 标记 AI 段落为陶土橙下划线
      renderAIMarkedParagraphs()

    case "review":
      // 审阅模式：人类 + AI 均可提交批注
      disableTextEditing()
      toolbar = [批注, 已解决]
      showAnnotations = true
      renderAnnotations()
```

## 4. AI 生成内容标记（核心）

```
function renderAIMarkedParagraphs():
  for each paragraph in paragraphs:
    if paragraph.aiMarked:
      // 视觉：陶土橙下划线 + 浅沙底纹
      css = {
        border-bottom: "2px solid #C97C3E"     // 陶土橙下划线
        background: "rgba(201, 124, 62, 0.06)"  // 极浅沙底纹
        position: relative
      }
      // hover 气泡提示
      onHover = showTooltip("🤖 文书成 · 修改 · 可撤销")

      // 撤销按钮（仅人类可见）
      showUndoButton(paragraph.id, onClick=undoAIMark)

    if paragraph.revision:
      // 有修改历史
      showRevisionBadge(paragraph.revision)
```

### 撤销 AI 修改

```
function undoAIMark(paragraphId):
  paragraph = findById(paragraphId)
  // 回退到 AI 修改前的文本
  if paragraph.revision.oldText:
    paragraph.text = paragraph.revision.oldText
  paragraph.aiMarked = false
  paragraph.revision = null
  paragraph.author = "human"

  // 通知后端：撤销此修改
  syncUndoToServer(paragraphId)

  // 更新文书成（AI）的上下文：下次不会再做此修改
  notifyHermes("undo_paragraph", { paragraphId, reason: "人工撤销" })
```

## 6. AI 批注（审阅模式）

```
// Hermes 调用"文书成"专家自动审阅
function aiReview(docId):
  // 1. 文书成（AI）通读全文
  findings = callHermesSkill("doc-review", docId)

  for each finding in findings:
    annotation = {
      id: uuid(),
      paragraphId: finding.paragraphId,
      author: "文书成（AI）",           // AI 身份标识
      authorType: "ai",                 // 区别人类
      text: finding.suggestion,         // e.g. "法条引用格式建议：'第X条第X款' → '第X条第X款第X项'"
      time: now(),
      status: "open",
      replies: []
    }
    annotations.push(annotation)
    syncToServer("add_annotation", annotation)

  // 2. 通知前端渲染
  pushEvent("annotations_updated", { count: findings.length })
```

## 7. 批注回复

```
function replyToAnnotation(annotationId, replyText):
  annotation = findAnnotation(annotationId)

  reply = {
    author: currentUser.name,
    authorType: currentUser.type,    // "human" | "ai"
    text: replyText,
    time: now()
  }
  annotation.replies.push(reply)

  // 如果标记"已解决"
  if replyText.includes("已解决") or replyText.includes("✓"):
    annotation.status = "resolved"
    dimParagraph(annotation.paragraphId)  // 段落灰蓝高亮消失

  syncToServer("reply_annotation", { annotationId, reply })
```

## 9. 协作锁（避免冲突覆盖）

```
function lockParagraph(paragraphId):
  // 同一段落同时只有一方可编辑
  paragraph = findById(paragraphId)
  if paragraph.locked:
    if paragraph.lockedBy == currentUser.type:
      return true   // 自己锁的
    else:
      showWarning("此段落正在被另一方编辑，请稍候")
      return false

  paragraph.locked = true
  paragraph.lockedBy = currentUser.type
  syncToServer("lock", { paragraphId, by: currentUser.type })

  // 30s 无操作自动释放锁
  startLockTimer(paragraphId, 30000)

function unlockParagraph(paragraphId):
  paragraph = findById(paragraphId)
  paragraph.locked = false
  paragraph.lockedBy = null
  syncToServer("unlock", { paragraphId })

function releaseLockOnIdle(paragraphId, idleMs=5000):
  // 5s 无操作自动释放
  onUserStopTyping(paragraphId, () => {
    unlockParagraph(paragraphId)
  })
```

## 10. 与 Hermes 基座通信

```
// 通过 eco-bridge → Hermes agent 通道
function syncToServer(action, data):
  payload = {
    docId: currentDoc.id,
    action: action,            // "add_annotation" | "reply_annotation" | "lock" | "undo_paragraph"
    data: data,
    timestamp: now(),
    user: currentUser
  }

  // 使用现有 agent 调用通道
  bridge.post("/api/office/sync", payload)
    .then(response => handleSyncResponse(response))
    .catch(err => queueForRetry(payload))  // 离线队列，联网后批量同步

function handleSyncResponse(response):
  if response.conflict:
    // 冲突处理
    showConflictWarning(response.conflictDetails)
    reloadDocument()  // 重新加载服务端版本
  elif response.aiResponse:
    // AI 有新的批注或修改
    mergeAIResponse(response.aiResponse)
```
